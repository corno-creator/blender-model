import bpy # ======================================== # 設定 # ======================================== EMITTER_NAME = "球" SOURCE_NAME = "Sphere" COLLECTION_NAME = "Particle_Converted" START_FRAME = 1 END_FRAME = 400 # Trueなら位置・回転・サイズをキーフレーム化 KEY_LOCATION = True KEY_ROTATION = True KEY_SCALE = True # 補間方法 USE_LINEAR = True # ======================================== # コレクション作成 # ======================================== def get_or_create_collection(): collection = bpy.data.collections.get(COLLECTION_NAME) if collection is None: collection = bpy.data.collections.new( COLLECTION_NAME ) bpy.context.scene.collection.children.link( collection ) return collection # ======================================== # 以前作ったParticleオブジェクトを削除 # ======================================== def clear_collection(collection): for obj in list(collection.objects): bpy.data.objects.remove( obj, do_unlink=True ) # ======================================== # Blender 5.x # ActionからFCurveを取得 # ======================================== def get_action_fcurves(action, anim_data): fcurves = [] if action is None: return fcurves if anim_data is None: return fcurves # Blender 5.xでは # # Action # └ Layers # └ Strips # └ Channelbag # └ FCurves # # という構造になっている for layer in action.layers: for strip in layer.strips: if strip.type != 'KEYFRAME': continue try: channelbag = strip.channelbag( anim_data.action_slot ) except Exception: channelbag = None if channelbag is None: continue for fcurve in channelbag.fcurves: fcurves.append(fcurve) return fcurves # ======================================== # Particle用オブジェクト作成 # ======================================== def create_particle_objects( particle_count, source_obj, collection ): objects = [] mesh = source_obj.data source_scale = source_obj.scale.copy() print("") print("元オブジェクト:", source_obj.name) print( "元Scale:", source_scale.x, source_scale.y, source_scale.z ) print( "作成するParticleオブジェクト数:", particle_count ) for i in range(particle_count): obj_name = "particle.{:03d}".format(i) obj = bpy.data.objects.new( obj_name, mesh ) collection.objects.link(obj) obj.rotation_mode = 'QUATERNION' objects.append(obj) return objects, source_scale # ======================================== # ParticleのTransformを設定 # ======================================== def set_transform( particle, obj, source_scale ): # ------------------------------------ # Location # ------------------------------------ obj.location = particle.location # ------------------------------------ # Rotation # ------------------------------------ obj.rotation_mode = 'QUATERNION' obj.rotation_quaternion = particle.rotation # ------------------------------------ # Scale # ------------------------------------ size = particle.size obj.scale = ( source_scale.x * size, source_scale.y * size, source_scale.z * size ) # ======================================== # キーフレーム作成 # ======================================== def insert_keyframes(obj): if KEY_LOCATION: obj.keyframe_insert( data_path="location" ) if KEY_ROTATION: obj.keyframe_insert( data_path="rotation_quaternion" ) if KEY_SCALE: obj.keyframe_insert( data_path="scale" ) # ======================================== # キーフレーム補間 # ======================================== def set_interpolation(obj): if not USE_LINEAR: return if obj.animation_data is None: return if obj.animation_data.action is None: return action = obj.animation_data.action fcurves = get_action_fcurves( action, obj.animation_data ) for fcurve in fcurves: for keyframe in fcurve.keyframe_points: keyframe.interpolation = 'LINEAR' # ======================================== # メイン処理 # ======================================== def main(): print("") print("========================================") print("Particle → Keyframe") print("Blender 5.2対応") print("========================================") scene = bpy.context.scene # ------------------------------------ # 発生元取得 # ------------------------------------ emitter = bpy.data.objects.get( EMITTER_NAME ) if emitter is None: print("") print("ERROR") print( "発生元「{}」が見つかりません".format( EMITTER_NAME ) ) return print("") print("発生元:", emitter.name) # ------------------------------------ # Particle System確認 # ------------------------------------ if len(emitter.particle_systems) == 0: print("") print("ERROR") print("Particle Systemがありません") return print( "Particle System数:", len(emitter.particle_systems) ) # ------------------------------------ # 元オブジェクト取得 # ------------------------------------ source_obj = bpy.data.objects.get( SOURCE_NAME ) if source_obj is None: print("") print("ERROR") print( "Particle表示用オブジェクト「{}」が見つかりません".format( SOURCE_NAME ) ) return print( "Particle表示用オブジェクト:", source_obj.name ) # ------------------------------------ # 元Scale # ------------------------------------ source_scale = source_obj.scale.copy() print("") print("元オブジェクトScale:") print("X:", source_scale.x) print("Y:", source_scale.y) print("Z:", source_scale.z) # ------------------------------------ # コレクション # ------------------------------------ collection = get_or_create_collection() clear_collection(collection) # ------------------------------------ # 評価済みParticle取得 # ------------------------------------ depsgraph = bpy.context.evaluated_depsgraph_get() # ------------------------------------ # Particle数取得 # ------------------------------------ evaluated_emitter = emitter.evaluated_get( depsgraph ) evaluated_ps = ( evaluated_emitter.particle_systems[0] ) particle_count = len( evaluated_ps.particles ) print("") print( "評価済みParticle数:", particle_count ) if particle_count == 0: print("") print("ERROR") print("評価済みParticleが0個です") return # ------------------------------------ # Particleオブジェクト作成 # ------------------------------------ particle_objects, source_scale = ( create_particle_objects( particle_count, source_obj, collection ) ) print("") print( "生成オブジェクト数:", len(particle_objects) ) # ------------------------------------ # フレーム範囲 # ------------------------------------ start_frame = START_FRAME end_frame = END_FRAME print("") print( "フレーム:", start_frame, "～", end_frame ) print("") print("========================================") print("キーフレーム作成開始") print("========================================") # ==================================== # フレームごとの処理 # ==================================== for frame in range( start_frame, end_frame + 1 ): # -------------------------------- # フレーム変更 # -------------------------------- scene.frame_set(frame) # -------------------------------- # Dependency Graph更新 # -------------------------------- bpy.context.view_layer.update() # -------------------------------- # 評価済みEmitter取得 # -------------------------------- depsgraph = ( bpy.context.evaluated_depsgraph_get() ) evaluated_emitter = emitter.evaluated_get( depsgraph ) evaluated_ps = ( evaluated_emitter.particle_systems[0] ) particles = evaluated_ps.particles # -------------------------------- # Particle数確認 # -------------------------------- count = min( len(particles), len(particle_objects) ) # -------------------------------- # Particleごとの処理 # -------------------------------- for i in range(count): particle = particles[i] obj = particle_objects[i] # ---------------------------- # Particleが生存している場合 # ---------------------------- if particle.alive_state == 'ALIVE': set_transform( particle, obj, source_scale ) insert_keyframes(obj) else: # ------------------------ # 生存していないParticle # ------------------------ obj.scale = ( 0.0, 0.0, 0.0 ) insert_keyframes(obj) # -------------------------------- # 進捗表示 # -------------------------------- if ( frame == start_frame or frame == end_frame or frame % 20 == 0 ): print( "Frame {} / {} Particle {}".format( frame, end_frame, count ) ) # ==================================== # 補間設定 # ==================================== print("") print("キーフレーム補間設定中...") for obj in particle_objects: set_interpolation(obj) # ==================================== # 最初のフレームへ戻す # ==================================== scene.frame_set(start_frame) bpy.context.view_layer.update() # ==================================== # 結果確認 # ==================================== print("") print("========================================") print("キーフレーム確認") print("========================================") success_count = 0 for obj in particle_objects: if ( obj.animation_data is not None and obj.animation_data.action is not None ): total_keys = 0 fcurves = get_action_fcurves( obj.animation_data.action, obj.animation_data ) for fcurve in fcurves: total_keys += len( fcurve.keyframe_points ) if total_keys > 0: success_count += 1 print( "キーフレームあり:", success_count, "/", len(particle_objects) ) print("") print("========================================") print("処理完了") print("========================================") print( "変換先コレクション:", COLLECTION_NAME ) print( "オブジェクト数:", len(particle_objects) ) print( "Particle System元:", emitter.name ) print("") print("コレカには変更を加えていません。") print("========================================") # ======================================== # 実行 # ======================================== main()
